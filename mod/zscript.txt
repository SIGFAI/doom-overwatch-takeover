// Roadhog: chain hook pulls you in, then he smashes you (replaces the demon)
class OWRoadhog : OWHero replaces Demon
{
	Default
	{
		Health 200; Speed 8; PainChance 70; Mass 400;
		Radius 34; Height 86;
		MeleeRange 80;
		SeeSound "ow/v/hog"; PainSound "demon/pain"; DeathSound "demon/death"; ActiveSound "demon/active";
		Obituary "%o was hooked and gutted by %k.";
		Tag "Roadhog";
		MinMissileChance 120;
	}
	// one tic of pulling: a chain of links from the hog to the victim, and the victim slides towards the hog
	void A_HogPull()
	{
		if (!target) return;
		Vector3 d = Vec3To(target);
		double dist = d.Length();
		if (dist < 1) return;
		Vector3 n = d / dist;
		for (double i = 24; i < dist; i += 18)
			Spawn("OWChain", pos + (0, 0, 54) + n * i);
		Spawn("OWHookHead", target.pos + (0, 0, target.height * 0.6));
		if (dist > 150)
		{
			target.Vel.X = -n.X * 22;
			target.Vel.Y = -n.Y * 22;
		}
	}

	{
		A_FaceTarget();
		A_StartSound("ow/hook", CHAN_WEAPON);

		if (h) h.Feed("ROADHOG  hooked  " .. ((target && target.player) ? "you" : target.GetTag()));
	}

// ---------- rules, HUD, announcer ----------

class OWHandler : EventHandler
{
	int ult;            // 0..100
	int streak, streakTime;
	int introTime;
	int potgTime, ultReadyTime;
	Array<String> feedLines;
	Array<int> feedTime;

	void Feed(String s)
	{
		feedLines.Push(s);
		feedTime.Push(level.time);
		if (feedLines.Size() > 5) { feedLines.Delete(0); feedTime.Delete(0); }
	}

	override void PlayerEntered(PlayerEvent e)
	{
		let p = players[e.PlayerNumber].mo;
		if (!p) return;
		if (!p.FindInventory("OWPulsePistols")) p.GiveInventory("OWPulsePistols", 1);
		let w = Weapon(p.FindInventory("OWPulsePistols"));
		if (w) players[e.PlayerNumber].PendingWeapon = w;
		introTime = level.time;
		S_StartSound("ow/v/intro", CHAN_AUTO, 0, 1.0, ATTN_NONE);
	}

	void FireUlt(Actor p)
	{
		if (!p || ult < 100) return;
		ult = 0;
		p.A_StartSound("ow/v/ult", CHAN_VOICE);
		let b = p.SpawnPlayerMissile("OWPulseBomb");
		// lock onto the closest hero in view
		Actor best = null;
		double bd = 1800;
		let it = ThinkerIterator.Create("OWHero");
		Actor h;
		while (h = Actor(it.Next()))
		{
			if (h.health <= 0 || !p.CheckSight(h)) continue;
			double d = p.Distance3D(h);
			if (d < bd) { bd = d; best = h; }
		}
		if (b && best)
		{
			b.tracer = best;
			Vector3 dir = (best.pos + (0, 0, best.height * 0.5) - b.pos).Unit();
			b.Vel = dir * b.Speed;
			b.angle = atan2(dir.y, dir.x);
		}
		Feed("PULSE BOMB  launched");
	}

	// ---- scripted showcase (set by demo.cfg: set ow_demo 1) ----
	bool demoGod;

	void DemoSpawn(Class<Actor> cls, double side)
	{
		let p = players[0].mo;
		if (!p) return;
		for (double d = 170; d >= 70; d -= 25)
		{
			Vector3 at = p.Vec3Angle(d, p.angle + side, 0);
			let m = Actor.Spawn(cls, at);
			if (!m) return;
			if (m.TestMobjLocation())
			{
				m.angle = p.angle + 180;
				return;
			}
			m.Destroy();
		}
	}

	override void WorldTick()
	{
		let cv = CVar.FindCVar("ow_demo");
		if (!cv || cv.GetInt() == 0) return;
		let p = players[0].mo;
		if (!p) return;
		if (!demoGod) { demoGod = true; p.player.cheats |= CF_GODMODE; }
		int t = level.time;
		// the script repeats its second half every 35 s after the first 40 s
		int at = t;
		if (at >= 35 * 40) at = 35 * 5 + (at - 35 * 40) % (35 * 35);
		switch (at)
		{
		case 10:      DemoSpawn("OWReaper", -30); DemoSpawn("OWSoldier", 30); DemoSpawn("OWSoldier", 0); break;
		case 35 * 5:  DemoSpawn("OWRoadhog", 0); DemoSpawn("OWSoldier", -35); break;
		case 35 * 10: DemoSpawn("OWMercy", 25); DemoSpawn("OWReaper", -25); break;
		case 35 * 14: DemoSpawn("OWBastion", 0); DemoSpawn("OWSoldier", 35); DemoSpawn("OWReaper", -35); break;
		case 35 * 18: ult = 100; ultReadyTime = level.time; S_StartSound("ow/v/ult", CHAN_AUTO, 0, 1.0, ATTN_NONE); break;
		case 35 * 20: FireUlt(p); break;
		case 35 * 25: DemoSpawn("OWRoadhog", -20); DemoSpawn("OWMercy", 20); DemoSpawn("OWSoldier", 0); break;
		case 35 * 30: DemoSpawn("OWBastion", 20); DemoSpawn("OWReaper", -20); DemoSpawn("OWSoldier", 40); break;
		case 35 * 35: DemoSpawn("OWRoadhog", 30); DemoSpawn("OWSoldier", -30); break;
		}
	}

	override void WorldThingDied(WorldEvent e)
	{
		let t = e.Thing;
		if (!(t is "OWHero")) return;
		Feed("ELIMINATED  " .. t.GetTag());
		int before = ult;
		ult = min(100, ult + ((t.SpawnHealth() >= 150) ? 20 : 8));
		if (before < 100 && ult >= 100)
		{
			ultReadyTime = level.time;
			S_StartSound("ow/v/ult", CHAN_AUTO, 0, 1.0, ATTN_NONE);
		}
		if (level.time - streakTime > 35 * 7) streak = 0;
		streak++;
		streakTime = level.time;
		if (streak == 5 && (potgTime == 0 || level.time - potgTime > 35 * 20))
		{
			potgTime = level.time;
			S_StartSound("ow/v/potg", CHAN_AUTO, 0, 1.0, ATTN_NONE);
		}
	}

	override void WorldThingRevived(WorldEvent e)
	{
		if (e.Thing is "OWHero") Feed("MERCY  resurrected  " .. e.Thing.GetTag());
	}

	override void NetworkProcess(ConsoleEvent e)
	{
		if (e.Name == "ow_charge") { ult = 100; ultReadyTime = level.time; return; }
		if (e.Name == "ow_ult")
		{
			FireUlt(players[e.Player].mo);
		}
	}

	override void RenderOverlay(RenderEvent e)
	{
		int sw = Screen.GetWidth(), sh = Screen.GetHeight();
		double sc = sh / 200.0;
		int now = level.time;

		// intro banner
		int age = now - introTime;
		if (age < 35 * 5)
		{
			double a = age > 35 * 4 ? (35 * 5 - age) / 35.0 : 1.0;
			String t1 = "OVERWATCH TAKEOVER";
			String t2 = "Eliminate every hero";
			Screen.Dim(0x000010, 0.5 * a, 0, int(30 * sc), sw, int(48 * sc));
			Screen.DrawText(BigFont, Font.CR_ORANGE, (240 - BigFont.StringWidth(t1)) / 2, 26, t1, DTA_VirtualWidth, 240, DTA_VirtualHeight, 150, DTA_Alpha, a);
			Screen.DrawText(SmallFont, Font.CR_WHITE, (320 - SmallFont.StringWidth(t2)) / 2, 62, t2, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200, DTA_Alpha, a);
		}

		// play of the game
		age = now - potgTime;
		if (potgTime > 0 && age < 35 * 4)
		{
			String t = "PLAY OF THE GAME";
			Screen.Dim(0x101830, 0.6, 0, int(62 * sc), sw, int(44 * sc));
			Screen.DrawText(BigFont, (age & 8) ? Font.CR_GOLD : Font.CR_YELLOW, (240 - BigFont.StringWidth(t)) / 2, 54, t, DTA_VirtualWidth, 240, DTA_VirtualHeight, 150);
		}

		// ultimate meter
		int bx = int(8 * sc), by = int(18 * sc), bw = int(70 * sc), bh = int(8 * sc);
		bool ready = ult >= 100;
		Screen.Dim(0x000000, 0.6, bx - 2, by - 2, bw + 4, bh + 4);
		Screen.Dim(ready ? ((now & 8) ? 0xFFE060 : 0xFFA020) : 0x40B0FF, 0.9, bx, by, int(bw * ult / 100.0), bh);
		String ut = ready ? "ULTIMATE READY - PRESS U" : String.Format("ULTIMATE %d%%", ult);
		Screen.DrawText(SmallFont, ready ? Font.CR_GOLD : Font.CR_CYAN, 8, 6, ut, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200);

		// kill feed
		for (int i = 0; i < feedLines.Size(); i++)
		{
			int fa = now - feedTime[i];
			if (fa > 35 * 6) continue;
			String s = feedLines[i];
			int w = SmallFont.StringWidth(s);
			Screen.Dim(0x000000, 0.55, int((314 - w - 3) * sc), int((8 + i * 11) * sc), int((w + 6) * sc), int(10 * sc));
			Screen.DrawText(SmallFont, s.Left(5) == "MERCY" ? Font.CR_GOLD : (s.Left(4) == "PULS" ? Font.CR_CYAN : Font.CR_RED), 314 - w, 9 + i * 11, s, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200);
		}
	}
}
